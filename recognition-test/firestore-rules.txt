// ============================================================
// قواعد Firestore لاختبار التعرّف (recognition-test)
// ============================================================
// أين تُلصق؟  Firebase Console ← Firestore Database ← Rules
// الصق الكتلة كلها داخل:
//     match /databases/{database}/documents {
//        ... هنا ...
//     }
// بجانب القواعد الموجودة، ولا تحذف شيئًا منها. ثم اضغط Publish.
//
// تحذير مهم: لو في قواعدك الحالية سطر عام زي
//     match /{document=**} { allow read, write: if true; }
// أو  allow read: if true;  على كل شيء، فهذه القواعد لن تحمي شيئًا،
// لأن Firestore يسمح لو أي قاعدة سمحت. وقتها المفتاح والأسماء
// تبقى مقروءة لأي حد. ابعت القواعد الحالية وأنا أظبطها لك.
// ============================================================

    function rtAdmin() {
      return request.auth != null && request.auth.token.email == 'admin@fouad-academy.com';
    }
    function rtSigned() {
      return request.auth != null;
    }
    function rtOwnId(rid) {
      return rtSigned() && rid.split('_')[0] == request.auth.uid;
    }

    // نصوص البطاقات الثلاث لكل رابط: العميل يقرأ رابطه بالرمز فقط (بلا list)
    match /rt_links/{token} {
      allow get: if rtSigned();
      allow list, write: if rtAdmin();

      // الأسماء: لا تُقرأ إلا بعد حفظ كلمة العميل في إجابته
      match /private/{doc} {
        allow get: if rtAdmin() || (
          rtSigned() && doc == 'names'
          && exists(/databases/$(database)/documents/rt_responses/$(request.auth.uid + '_' + token))
          && get(/databases/$(database)/documents/rt_responses/$(request.auth.uid + '_' + token)).data.get('ownWord', '') != ''
        );
        allow list, write: if rtAdmin();
      }
    }

    // الظروف: العميل يقرأ ظروف حرفه ويزيد العدّاد بواحد فقط (داخل معاملة)
    match /rt_envelopes/{letter} {
      allow get: if rtSigned();
      allow update: if rtAdmin() || (
        rtSigned()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['next'])
        && request.resource.data.next == resource.data.next + 1
      );
      allow list, create, delete: if rtAdmin();
    }

    // الإجابات: بلا اسم. كل عميل يكتب ويقرأ وثيقته فقط
    match /rt_responses/{rid} {
      allow get: if rtAdmin() || rtOwnId(rid);
      allow create: if rtOwnId(rid)
        && request.resource.data.uid == request.auth.uid
        && rid == request.auth.uid + '_' + request.resource.data.token
        && exists(/databases/$(database)/documents/rt_links/$(request.resource.data.token))
        && request.resource.data.letter == get(/databases/$(database)/documents/rt_links/$(request.resource.data.token)).data.letter;
      allow update: if rtAdmin() || (
        rtOwnId(rid)
        && request.resource.data.uid == resource.data.uid
        && request.resource.data.token == resource.data.token
        && request.resource.data.letter == resource.data.letter
      );
      allow list, delete: if rtAdmin();
    }

    // الأسماء ووسائل التواصل: العميل ينشئ فقط، والأدمن يقرأ
    match /rt_people/{rid} {
      allow create: if rtOwnId(rid);
      allow read, update, delete: if rtAdmin();
    }

    // المفتاح: الأدمن فقط
    match /rt_key/{doc} {
      allow read, write: if rtAdmin();
    }

    // وسيلة التواصل في شاشة الختام
    match /rt_settings/{doc} {
      allow get: if rtSigned();
      allow list, write: if rtAdmin();
    }
